1. 3D Binaural HRTF & Steam Audio Wall Occlusion

from pyaudiogaming.sound_pool import output, sound, update_hrtf_listener, hrtf_add_box_preset, hrtf_set_auto_raycast

out = output(device=-1, frequency=44100)
hrtf_set_auto_raycast(True)
hrtf_add_box_preset(1, min_x=4, max_x=5, min_y=-6, max_y=6, min_z=0, max_z=4)

s = sound()
s.stream("res/sounds/beacon.ogg", mono=True)
s.attach_hrtf_dsp()
s.play_looped()

update_hrtf_listener(0, 0, 1.7, 0, 1, 0, 0, 0, 1)

2. Encrypted SPAK Sound Archives

from pyaudiogaming.utils import make_spak, load_spak

make_spak("res/sounds", output_path="res/sounds.zb", password="pw", encrypt_bytes=256)
load_spak("res/sounds.zb", password="pw", mode="lazy", target="both")
